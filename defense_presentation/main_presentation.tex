\subsection{Effect of Reduction Ratio $R$ on Scan Spacing $S_{\text{scan}}$}
\begin{frame}
    \frametitle{Effect of Reduction Ratio $R$ on Scan Spacing $S_{\text{scan}}$}
    \begin{columns}
        \column{0.4\textwidth}
        \begin{equation*}
        S_{\text{scan}} = \varphi_{\text{step}}\cdot\sin(\theta)
        \end{equation*}
        \vspace{-0.5in}
            \begin{figure}
                \centering
                \input{figures/angleRes_tiltOnly.tex}
                \caption*{Without a gearbox}
            \end{figure}
        \column{0.55\textwidth}
        \begin{equation*}
         \varphi_{\text{eff}} = \frac{\varphi_{\text{step}}}{\color{cvut_navy}R},   S_{\text{scan}} = \varphi_{\text{eff}}\cdot\sin(\theta)
        \end{equation*}
            \begin{figure}
                \centering
                \includegraphics[width=\textwidth]{figures/angleRes_TiltPlusGearbox.png}
                \caption*{With a gearbox}
            \end{figure}
\end{columns}
\end{frame}

\section{Conclusion}
\begin{frame}{Conclusion}
A platform enabling 3D reconstruction with 2D LiDAR was built. It features a gearbox with a high reduction ratio, allowing for dense point clouds even at large tilt angles.\\\vspace{0.2cm}
Code for data acquisition, processing and 3D reconstruction visualisation was written.\\\vspace{0.2cm}
The prototype was tested by scanning a room, transformation corrections were applied.
\end{frame}

\begin{frame}{Odpovědi na otázky}
    
    {\color{yellow}Q3:} Jaká je časová náročnost provedení celého 3D scanu?\\
    {\color{cvut_navy}A3:} $450\,\text{kroků}\cdot0.105\,\text{s/krok} + 10\,\text{s(homing)} \approx60\,\text{s}.$
    Reálná doba je však o cca 10\,s delší kvůli přenosu dat přes Wi-Fi. V ojedinělých případech může dojít ke ztrátě spojení nebo komunikačním prodlevám, což může dále prodloužít dobu skenování.
    
\end{frame}
