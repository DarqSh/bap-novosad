\begin{frame}
\frametitle{Software}

\begin{center}
LiDAR $\rightarrow$ Pico 2 W $\rightarrow$ Wi-Fi $\rightarrow$ PC $\rightarrow$ Open3D / Flask
\end{center}

\begin{itemize}
    \item LiDAR posílá data přes UART
    \item Pico posílá data do PC přes Wi-Fi
    \item PC rekonstruuje point cloud a ukládá výsledky
    \item Flask dashboard zobrazuje stav skenování a poslední uložený scan
\end{itemize}
\end{frame}
